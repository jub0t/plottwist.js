Make it yours

Every look on this page is a set of theme tokens. Extend a preset, register your own, or override per chart.

// Presets: 'midnight' (default) | 'dark'. Canvases are transparent: the page owns the background.
new IsoBarChart(el, { data, theme: 'midnight', grid: 'none' });  // grid: 'dots' | 'lines' | 'none'

// Extend one and override any token
registerTheme('brand', {
  extends: 'midnight',
  series: ['#7c5cff', '#ff4f9a', '#2f7bff'],  // categorical, in fixed order
  scale: 'emerald',                           // or an array of stops, low → high
  glow: 0.6, pool: 0.4, gradient: 0.5, edge: 0.4,
  gridStyle: 'dots', floor: '#0c0c14',
  font: 'Inter, sans-serif', fontMono: 'JetBrains Mono, monospace',
});

// Pin colours to entities so they never shift when data changes
new IsoBarChart(el, { data, stack: 'channel', colors: { Online: '#7c5cff', Retail: '#ff4f9a' } });

// Switch at runtime
chart.setTheme('brand');
